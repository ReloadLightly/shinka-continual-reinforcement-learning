def update_sigma(sigma, stats, memory):
    mean = clip(stats[0], 0.0, 1.0)
    archive = clip(stats[3], 0.0, 1.0)
    quality = 0.55 * mean + 0.45 * archive
    previous = 0.55 * memory[0] + 0.45 * memory[1]
    change = quality - previous
    decline = maximum(previous - quality - 0.04, 0.0)
    shock = maximum(0.78 * memory[2], tanh(5.0 * decline))
    quiet = exp(-30.0 * abs(change))
    plateau = 0.88 * memory[3] + 0.12 * quiet
    headroom = 1.0 - quality
    target = 0.035 + 0.50 * headroom * headroom
    target = target + 0.25 * plateau * headroom + 0.70 * shock
    success = clip(stats[4], 0.0, 1.0)
    target = target * exp(0.30 * (success - 0.35))
    target = clip(target, 0.025, 1.20)
    rate = 0.18 + 0.32 * shock
    current = clip(sigma, 0.001, 2.0)
    next_sigma = exp((1.0 - rate) * log(current) + rate * log(target))
    next_mean = 0.75 * memory[0] + 0.25 * mean
    next_archive = 0.75 * memory[1] + 0.25 * archive
    next_memory = stack([next_mean, next_archive, shock, plateau])
    return next_sigma, next_memory
